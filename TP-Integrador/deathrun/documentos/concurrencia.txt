OBJETIVO:

class_name Future
extends RefCounted

signal completed(result)

var _result = null
var _is_completed: bool = false
var _error: String = ""
var _timeout_timer: Timer = null

func complete(result):
    """Completa el Future con un resultado exitoso"""
    if _is_completed:
        return
    
    _result = result
    _is_completed = true
    
    if _timeout_timer:
        _timeout_timer.stop()
        _timeout_timer.queue_free()
    
    completed.emit(result)

func fail(error_message: String):
    """Completa el Future con un error"""
    if _is_completed:
        return
    
    _error = error_message
    _is_completed = true
    
    if _timeout_timer:
        _timeout_timer.stop()
        _timeout_timer.queue_free()
    
    completed.emit(null)

func await_result(timeout_seconds: float = 5.0):
    """Espera el resultado con timeout"""
    if _is_completed:
        return _result
    
    # Configurar timeout si no está completado
    if timeout_seconds > 0:
        await _setup_timeout(timeout_seconds)
    
    # Esperar señal de completado
    await completed
    
    return _result

func _setup_timeout(seconds: float):
    """Configura un timer para timeout"""
    _timeout_timer = Timer.new()
    _timeout_timer.wait_time = seconds
    _timeout_timer.one_shot = true
    _timeout_timer.timeout.connect(func():
        fail("Timeout: no response after %.1f seconds" % seconds)
    )
    # Agregar a la raíz del árbol
    Engine.get_main_loop().root.add_child(_timeout_timer)
    _timeout_timer.start()

func is_completed() -> bool:
    return _is_completed

func has_error() -> bool:
    return _error != ""

func get_error() -> String:
    return _error

# MODIFICAR: Activación asíncrona con confirmación
func activate_selected_trap_async() -> Dictionary:
    """
    Activa la trampa seleccionada y espera confirmación del servidor.
    Retorna: { "success": bool, "message": String }
    """
    
    if available_traps.size() == 0:
        return {"success": false, "message": "No hay trampas disponibles"}
    
    var trap_id = available_traps[selected_trap_index]
    var request_id = request_id_counter
    request_id_counter += 1
    
    print("[CLIENTE] Solicitando activación de trampa %d (request: %d)" % [
        trap_id, 
        request_id
    ])
    
    # Crear Future para esta solicitud
    var future = Future.new()
    pending_requests[request_id] = future
    
    # Enviar solicitud al servidor
    if NetworkManager.is_multiplayer_active():
        var server_trap_manager = get_node_or_null("/root/Level01/TrapManager")
        
        if server_trap_manager:
            server_trap_manager.rpc_id(
                1,  # ID del servidor
                "request_trap_activation",
                trap_id,
                request_id
            )
        else:
            future.fail("Error: No se encontró TrapManager")
            return {"success": false, "message": "Error interno"}
    else:
        # Modo local (sin red)
        var success = trap_manager.activate_trap(trap_id)
        future.complete({"success": success, "message": "Modo local"})
    
    # ESPERAR respuesta con timeout de 5 segundos
    var result = await future.await_result(5.0)
    
    # Limpiar
    pending_requests.erase(request_id)
    
    # Procesar resultado
    if future.has_error():
        print("[CLIENTE] Error: %s" % future.get_error())
        return {"success": false, "message": future.get_error()}
    
    if result == null:
        return {"success": false, "message": "Sin respuesta del servidor"}
    
    print("[CLIENTE] Respuesta recibida: %s" % result.message)
    return result

OBJETIVO:
---------
Separar el procesamiento de red del hilo principal del juego en el servidor.
El hilo principal maneja física, rendering, lógica del juego.
El worker thread maneja envío/recepción de paquetes de red.

ARQUITECTURA PROPUESTA:
----------------------

SERVIDOR (Host)
═══════════════════════════════════════════════════════════════

┌─────────────────────────────────────────────────────────────┐
│                    MAIN THREAD                              │
│  ┌───────────────────────────────────────────────────────┐  │
│  │ Game Logic (60 FPS)                                   │  │
│  │ - Player physics                                      │  │
│  │ - Trap state machines                                 │  │
│  │ - Collision detection                                 │  │
│  │ - Scene Tree updates                                  │  │
│  └────────────┬──────────────────────────────────────────┘  │
│               │                                             │
│               ▼                                             │
│  ┌───────────────────────────────────────────────────────┐  │
│  │ Outgoing Message Queue (Thread-Safe)                  │  │
│  │ [msg1, msg2, msg3, ...]                               │  │
│  └────────────┬──────────────────────────────────────────┘  │
└───────────────┼─────────────────────────────────────────────┘
                │
                │ Mutex protected
                │
┌───────────────▼─────────────────────────────────────────────┐
│                  NETWORK WORKER THREAD                      │
│  ┌───────────────────────────────────────────────────────┐  │
│  │ Network I/O Loop                                      │  │
│  │                                                       │  │
│  │ while running:                                        │  │
│  │   1. Poll incoming packets                            │  │
│  │   2. Push to incoming_queue (mutex)                   │  │
│  │   3. Pop from outgoing_queue (mutex)                  │  │
│  │   4. Send packets via ENet                            │  │
│  │   5. Sleep 1ms (no busy-wait)                         │  │
│  └───────────────────────────────────────────────────────┘  │
│               │                                             │
│               ▼                                             │
│  ┌───────────────────────────────────────────────────────┐  │
│  │ Incoming Message Queue (Thread-Safe)                  │  │
│  │ [msg1, msg2, msg3, ...]                               │  │
│  └────────────┬──────────────────────────────────────────┘  │
└───────────────┼─────────────────────────────────────────────┘

class_name NetworkWorker
extends RefCounted

# Thread y sincronización
var worker_thread: Thread = null
var mutex: Mutex = Mutex.new()
var semaphore: Semaphore = Semaphore.new()
var should_exit: bool = false

# Colas de mensajes (protegidas por mutex)
var outgoing_queue: Array[NetworkMessage] = []
var incoming_queue: Array[NetworkMessage] = []

# Estadísticas
var stats = {
    "messages_sent": 0,
    "messages_received": 0,
    "bytes_sent": 0,
    "bytes_received": 0,
    "queue_overflows": 0
}

const MAX_QUEUE_SIZE = 1000  # Prevenir memory leaks

# Iniciar worker thread
func start():
    if worker_thread != null:
        push_warning("Worker thread ya está corriendo")
        return false
    
    should_exit = false
    worker_thread = Thread.new()
    worker_thread.start(_worker_loop)
    
    print("[NetworkWorker] Thread iniciado")
    return true

# Detener worker thread
func stop():
    if worker_thread == null:
        return
    
    print("[NetworkWorker] Deteniendo thread...")
    should_exit = true
    semaphore.post()  # Despertar thread si está bloqueado
    
    worker_thread.wait_to_finish()
    worker_thread = null
    
    print("[NetworkWorker] Thread detenido")

# Enviar mensaje (thread-safe)
func queue_message(msg: NetworkMessage):
    mutex.lock()
    
    if outgoing_queue.size() >= MAX_QUEUE_SIZE:
        stats.queue_overflows += 1
        push_warning("Cola de salida llena, descartando mensaje")
        mutex.unlock()
        return
    
    outgoing_queue.append(msg)
    mutex.unlock()
    
    # Notificar al worker thread
    semaphore.post()

# Obtener mensajes recibidos (thread-safe)
func pop_incoming_messages() -> Array[NetworkMessage]:
    mutex.lock()
    var messages = incoming_queue.duplicate()
    incoming_queue.clear()
    mutex.unlock()
    
    return messages

# === WORKER THREAD LOOP (corre en thread separado) ===
func _worker_loop():
    print("[NetworkWorker] Loop iniciado en thread ID: ", OS.get_thread_caller_id())
    
    while not should_exit:
        # 1. Procesar mensajes salientes
        _process_outgoing_messages()
        
        # 2. Recibir mensajes entrantes (simulado - en realidad ENet maneja esto)
        _process_incoming_packets()
        
        # 3. Dormir para no consumir 100% CPU
        OS.delay_msec(1)  # 1ms de sleep (1000 Hz máximo)
    
    print("[NetworkWorker] Loop terminado")

func _process_outgoing_messages():
    """Envía mensajes de la cola de salida"""
    
    # Obtener batch de mensajes (para reducir locks)
    mutex.lock()
    if outgoing_queue.is_empty():
        mutex.unlock()
        return
    
    # Copiar hasta 10 mensajes
    var batch_size = min(10, outgoing_queue.size())
    var messages = outgoing_queue.slice(0, batch_size)
    outgoing_queue = outgoing_queue.slice(batch_size)
    mutex.unlock()
    
    # Procesar fuera del lock (para no bloquear el main thread)
    for msg in messages:
        _send_message(msg)

    var size = _estimate_message_size(msg)
    
    mutex.lock()
    stats.messages_sent += 1
    stats.bytes_sent += size
    mutex.unlock()

    """Recibe paquetes de red (simulado)"""

func _queue_incoming_message(msg: NetworkMessage):
    """Agrega mensaje a la cola de entrada (thread-safe)"""
    
    mutex.lock()
    
    if incoming_queue.size() >= MAX_QUEUE_SIZE:
        stats.queue_overflows += 1
        push_warning("Cola de entrada llena")
        mutex.unlock()
        return
    
    incoming_queue.append(msg)
    stats.messages_received += 1
    mutex.unlock()

func _estimate_message_size(msg: NetworkMessage) -> int:
    """Estima el tamaño de un mensaje en bytes"""
    var size = 8  # Header base
    size += msg.method_name.length()
    size += msg.args.size() * 8  # Estimación
    return size

func get_stats() -> Dictionary:
    mutex.lock()
    var stats_copy = stats.duplicate()
    mutex.unlock()
    return stats_copy

    """Detiene el worker thread"""
    if network_worker == null:
        return
    
    network_worker.stop()
    network_worker = null
    print("[NetworkManager] Worker thread detenido")

func _process(_delta):
    """Procesa mensajes del worker thread en el main thread"""
    if network_worker == null or not is_server():
        return
    
    # Obtener mensajes recibidos del worker thread
    var incoming = network_worker.pop_incoming_messages()
    
    for msg in incoming:
        _process_network_message(msg)

func _process_network_message(msg: NetworkMessage):
    """Procesa un mensaje de red (corre en main thread)"""
    match msg.type:
        NetworkMessage.Type.RPC_CALL:
            # Ejecutar RPC en el main thread
            if has_method(msg.method_name):
                callv(msg.method_name, msg.args)
        
        NetworkMessage.Type.TRAP_ACTIVATION:
            # Notificar al TrapManager
            var trap_manager = get_node_or_null("/root/Level01/TrapManager")
            if trap_manager:
                trap_manager.callv(msg.method_name, msg.args)

func disconnect_from_game():
    """Desconecta del juego multijugador"""
    
    # NUEVO: Detener worker thread
    _stop_network_worker()
    
    if peer:
        peer.close()
    multiplayer.multiplayer_peer = null
    is_host = false
    player_role = ""

5.1 ASYNC/AWAIT (Concurrencia Cooperativa)
------------------------------------------

- Worker thread pattern

- Separación de I/O y lógica del juego
- Thread safety

Archivos: network_worker.gd
Líneas clave:
  worker_thread = Thread.new()

------------------------------------
- Mutex para proteger estado compartido
- Critical sections
- Lock/Unlock pattern
- Deadlock avoidance (locks cortos)

Archivos: network_worker.gd
Líneas clave:
  mutex.lock()
  outgoing_queue.append(msg)
  mutex.unlock()

5.4 COMUNICACIÓN ENTRE THREADS

- Thread-safe queues
- Semaphore para señalización

Archivos: network_worker.gd
Líneas clave:

FLUJO TEMPORAL DE ACTIVACIÓN DE TRAMPA:
========================================

T=0ms:    Cliente: Usuario presiona ESPACIO
T=1ms:    Cliente: activate_trap_async() crea Future
T=2ms:    Cliente: RPC enviado (TCP) → Servidor
T=50ms:   Servidor: WORKER THREAD recibe paquete
T=51ms:   Servidor: MAIN THREAD procesa request_trap_activation()
T=52ms:   Servidor: TrapManager.activate_trap() ejecuta
T=53ms:   Servidor: SpikeTrap cambia estado → ACTIVATING
T=54ms:   Servidor: RPC response enviado (TCP) → Cliente
T=100ms:  Cliente: WORKER THREAD recibe respuesta
T=101ms:  Cliente: Future.complete() notifica
T=102ms:  Cliente: await retorna resultado
T=103ms:  Cliente: UI actualizada con ✅ o ❌

Total: ~100ms de latencia (depende de red)
